const emptyAddBg = document.getElementById('emptyAddBg'); const emptyAddBody = document.getElementById('emptyAddBody'); const emptyAddClose = document.getElementById('emptyAddClose'); async function openEmptyAddChat() { emptyAddBg.classList.add('open'); emptyAddBody.innerHTML = '
⏳ Memuat daftar…
'; try { const list = await api('/api/chats/available'); if (!list.length) { emptyAddBody.innerHTML = `
🎉
Semua chat udah kamu tambahkan!
Gak ada yang tersisa di sini.
`; return; } emptyAddBody.innerHTML = ''; // Public chat dulu const publicChats = list.filter(x => x.isPublic); const dmChats = list.filter(x => !x.isPublic); if (publicChats.length) { const sectionTitle = document.createElement('div'); sectionTitle.style.cssText = 'font-size:11.5px;font-weight:800;color:var(--blue-dark);text-transform:uppercase;letter-spacing:0.06em;margin:6px 0 8px;display:flex;align-items:center;gap:6px'; sectionTitle.innerHTML = '🌐 Public Chat'; emptyAddBody.appendChild(sectionTitle); publicChats.forEach(item => { emptyAddBody.appendChild(buildAddChatCard(item)); }); } if (dmChats.length) { const sectionTitle = document.createElement('div'); sectionTitle.style.cssText = 'font-size:11.5px;font-weight:800;color:var(--blue-dark);text-transform:uppercase;letter-spacing:0.06em;margin:14px 0 8px;display:flex;align-items:center;gap:6px'; sectionTitle.innerHTML = '👤 User Lain'; emptyAddBody.appendChild(sectionTitle); dmChats.forEach(item => { emptyAddBody.appendChild(buildAddChatCard(item)); }); } // Info footer const footer = document.createElement('div'); footer.style.cssText = 'text-align:center;font-size:11.5px;color:var(--muted);padding:14px 8px 4px;line-height:1.55;border-top:1px dashed rgba(255,255,255,0.5);margin-top:12px'; footer.innerHTML = '💡 Chat yang udah ditambahkan
gak akan muncul di sini lagi'; emptyAddBody.appendChild(footer); } catch (e) { emptyAddBody.innerHTML = `
❌ Gagal memuat: ${escapeHtml(e.message)}
`; } } function buildAddChatCard(item) { const el = document.createElement('div'); el.className = 'chat-item'; el.style.cursor = 'pointer'; el.style.marginBottom = '8px'; const avatarTxt = item.isPublic ? '🌐' : (item.name || '?')[0].toUpperCase(); const subText = item.subtitle || (item.isPublic ? 'Ngobrol bareng semua user' : 'Chat pribadi'); el.innerHTML = `
${avatarTxt}

${escapeHtml(item.name)}

${escapeHtml(subText)}
＋ Add
`; el.onclick = async () => { el.style.pointerEvents = 'none'; el.style.opacity = '0.6'; try { const payload = item.isPublic ? { type: 'group', chatId: item.id } : { type: 'dm', userId: item.userId, chatId: item.id }; const d = await api('/api/chats/add', { method: 'POST', body: JSON.stringify(payload) }); if (d.error) { alert('Gagal tambah: ' + d.error); el.style.pointerEvents = ''; el.style.opacity = ''; return; } // Refresh modal + chat list await loadChats(); // Item ini bakal ilang otomatis karena udah di-add openEmptyAddChat(); } catch (e) { alert('Gagal tambah: ' + e.message); el.style.pointerEvents = ''; el.style.opacity = ''; } }; return el; } emptyAddClose.onclick = () => emptyAddBg.classList.remove('open'); emptyAddBg.onclick = (e) => { if (e.target.id === 'emptyAddBg') emptyAddBg.classList.remove('open'); };
